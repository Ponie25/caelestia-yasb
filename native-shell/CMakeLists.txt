cmake_minimum_required(VERSION 3.24)
project(CaelestiaShell VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CAELESTIA_BUILD_GLASS "Rebuild the existing MIT Liquid Glass DLL (MSVC + Windows SDK)" ON)

add_library(caelestia_core STATIC src/core.cpp)
target_include_directories(caelestia_core PUBLIC include)
if(MSVC)
  target_compile_options(caelestia_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(caelestia_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(caelestia_core_tests tests/core_tests.cpp)
  target_link_libraries(caelestia_core_tests PRIVATE caelestia_core)
  target_compile_definitions(caelestia_core_tests PRIVATE CAELESTIA_CONFIG_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../config")
  add_test(NAME core COMMAND caelestia_core_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(caelestia-shell WIN32 src/windows_main.cpp src/render_device.cpp
    src/glass_bridge.cpp src/windows_services.cpp)
  target_compile_definitions(caelestia-shell PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(caelestia-shell PRIVATE caelestia_core d3d11 dxgi d2d1 dwrite dcomp
    dwmapi user32 gdi32 shell32 ole32 advapi32 shcore uuid)
  if(MSVC)
    target_sources(caelestia-shell PRIVATE src/app.manifest)
    target_compile_options(caelestia-shell PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_sources(caelestia-shell PRIVATE src/app.rc)
    target_compile_options(caelestia-shell PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(caelestia-shell PRIVATE -municode -static)
  endif()

  # Keep the renderer's source and ABI intact. No Qt/Python is required by this DLL.
  if(MSVC AND CAELESTIA_BUILD_GLASS)
    add_library(liquid_glass SHARED ../native/liquid-glass/lgwin.cpp)
    find_path(CAELESTIA_CPPWINRT winrt/Windows.Foundation.h PATHS
      "$ENV{WindowsSdkDir}/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt"
      "C:/Program Files (x86)/Windows Kits/10/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt")
    if(CAELESTIA_CPPWINRT)
      target_include_directories(liquid_glass PRIVATE "${CAELESTIA_CPPWINRT}")
    endif()
    target_compile_features(liquid_glass PRIVATE cxx_std_20)
    target_compile_options(liquid_glass PRIVATE /utf-8 /EHsc /W3)
    target_link_libraries(liquid_glass PRIVATE user32 d3d11 dxgi dcomp dwmapi d3dcompiler windowsapp)
    set_target_properties(liquid_glass PROPERTIES PREFIX "")
    add_dependencies(caelestia-shell liquid_glass)
    install(TARGETS liquid_glass RUNTIME DESTINATION .)
  elseif(CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
    # Cross-builds can package the already tracked x64 DLL, clearly identified in reports.
    add_custom_command(TARGET caelestia-shell POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/../config/dynamic-island/native/liquid_glass.dll"
      "$<TARGET_FILE_DIR:caelestia-shell>/liquid_glass.dll")
    install(FILES ../config/dynamic-island/native/liquid_glass.dll DESTINATION .)
  endif()
  install(TARGETS caelestia-shell RUNTIME DESTINATION .)
  install(FILES README.md DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs)
  install(DIRECTORY tools/ DESTINATION tools)
  install(DIRECTORY ../config/ DESTINATION config FILES_MATCHING PATTERN "*.json" PATTERN "*.ttf" PATTERN "*LICENSE*" PATTERN "OFL*" PATTERN "native" EXCLUDE)
  install(DIRECTORY ../config/theme-profiles/THIRD-PARTY/ DESTINATION config/theme-profiles/THIRD-PARTY)
  install(FILES ../LICENSE DESTINATION .)
  install(DIRECTORY ../licenses/ DESTINATION licenses)
  install(FILES ../native/liquid-glass/LICENSE DESTINATION licenses RENAME LiquidGlass-MIT.txt)
else()
  message(STATUS "macOS/Linux: building portable core/tests only; Windows GUI is not executed here.")
endif()
